\section{Contexto e problema}

\begin{frame}{O domínio: texto curto de varejo}
\begin{columns}[T]
\column{0.58\textwidth}
\begin{itemize}
  \item Descrições de produtos como saem do cupom fiscal:\\
        \texttt{CERV BRAHMA LT 350ML}, \texttt{SAB OMO 1KG}
  \item 4--50 caracteres, caixa alta, abreviação agressiva, sem contexto
  \item \textbf{621 categorias} com cauda longa severa (razão de
        desbalanceamento $>$ 1000:1)
  \item Aplicações: automação fiscal, \textit{pricing}, cesta de mercado,
        estatística de consumo
\end{itemize}
\column{0.42\textwidth}
\footnotesize
\begin{block}{Por que é difícil}
Pouca informação por instância + centenas de classes raras $\Rightarrow$
o Macro F1 é implacável e a partida a frio é crítica.
\end{block}
\end{columns}
\end{frame}
\note{[2:00] Ancore no exemplo concreto: leia \texttt{CERV BRAHMA LT 350ML} em voz alta. Ponto-chave: 621 classes, cauda longa, texto sem contexto. Feche com \"por isso o Macro F1 é implacável e a partida a frio é crítica\" --- planta as duas ideias que voltam.}


\begin{frame}{O gargalo real não é o modelo --- é o rótulo}
\begin{itemize}
  \item Classificadores modernos são bons; \textbf{obter rótulos} é o custo dominante
  \item \textbf{Aprendizado ativo (AA)}: o modelo escolhe o que vale a pena rotular
  \item Mas o laço clássico tem \textbf{duas fraturas}:
  \begin{enumerate}
    \item \emph{Partida a frio}: as estratégias dependem de um modelo que
          ainda não existe
    \item \emph{Oráculo perfeito}: assume-se anotador correto e de custo unitário
  \end{enumerate}
  \item \textbf{LLMs} mudam o jogo nas duas frentes --- podem \emph{selecionar}
        e podem \emph{rotular} --- mas trazem ruído estruturado e um novo
        modelo de custo (por token)
\end{itemize}
\vspace{0.4em}
\centering\small\color{ufprgreen}
É nessas duas fraturas que o FALCO atua.
\end{frame}
\note{[2:00] A frase que a banca precisa reter: \"o gargalo não é o modelo, é o rótulo\". Explique as duas fraturas devagar (cold start e oráculo perfeito). Feche dizendo que LLMs mexem nas duas --- transição natural para a hipótese.}


\begin{frame}{Pergunta e hipótese central (pré-registrada)}
\begin{block}{Pergunta}
Como reduzir o custo de rotulagem em classificação de texto curto em
português, usando LLMs nas fases em que o AA tradicional é mais frágil?
\end{block}
\begin{block}{Hipótese --- critério quantitativo fixado \emph{antes} dos experimentos}
Atingir \textbf{$\ge 95\%$} do Macro F1 da supervisão completa com
\textbf{$\le 30\%$} dos rótulos (via oráculo LLM), superando amostragem
aleatória e incerteza pura com significância estatística.
\end{block}
\footnotesize
Definido de antemão o que conta como \textbf{refutação}; \emph{gate}
pré-registrado de qualidade do oráculo. Postura: \textbf{nenhum número sem
artefato rastreável}.
\end{frame}
\note{[2:30] Este é o slide mais importante do início. Enfatize PRÉ-REGISTRO: o critério (95\%/30\%) e a definição de refutação foram fixados ANTES. Diga a frase-assinatura: \"nenhum número sem artefato rastreável\". A banca vai gostar disso.}

\section{Fundamentação}
\begin{frame}{Onde a tese se situa}
\begin{itemize}
  \item \textbf{Aprendizado ativo}: seleção por incerteza, comitê, densidade;
        limitações do laço (cold start, oráculo imperfeito, viés de amostragem)
  \item \textbf{LLM como oráculo}: Gilardi (2023) --- supera \textit{crowd};
        estudos de produto param em dezenas a $\approx$370 classes
  \item \textbf{Texto curto}: esparsidade, cauda longa, português de varejo
  \item \textbf{Instrumentação estatística}: Wilson, McNemar, Wilcoxon, bootstrap
\end{itemize}
\vspace{0.3em}
\begin{block}{Lacuna}
\footnotesize Ninguém combina cold start informado + oráculo LLM progressivo +
texto curto em português + custo instrumentado. É a célula vazia da tabela de lacunas.
\end{block}
\end{frame}
\note{[2:00] Não faça revisão de literatura falada; destaque só a LACUNA (a célula vazia: cold start informado + oráculo LLM + PT + custo instrumentado). Mencione que a instrumentação estatística está toda declarada --- Wilson, McNemar, Wilcoxon.}

\section{Resultados}
\begin{frame}{P1 --- A composição do conjunto inicial importa}
\begin{columns}[T]
\column{0.55\textwidth}
\begin{itemize}\footnotesize
  \item 47 tamanhos $\times$ 30 repetições
  \item Em $|L_0|=100$: acurácia varia \textbf{6,4 p.p.} só pelo sorteio
  \item A cobertura de classes é o mecanismo (Macro F1 segue a cobertura)
  \item Reexecução independente (protocolo com dedup): divergência $\le 0{,}7$ p.p.
\end{itemize}
\column{0.45\textwidth}
\footnotesize
\begin{block}{Consequência}
Se a composição importa tanto no regime pequeno --- o do cold start ---,
então \emph{construir} $L_0$ deliberadamente é a alavanca disponível antes
de existir qualquer rótulo.
\end{block}
\end{columns}
\end{frame}
\note{[2:30] O número que impressiona: 6,4 p.p. de variação SÓ pelo sorteio em L0=100. Explique o mecanismo (cobertura de classes). Feche: \"se a composição importa tanto, construir o L0 é a alavanca antes de existir rótulo\" --- prepara o P2.}

\begin{frame}{P3 --- A medição é parte do método (RQ4)}
\begin{itemize}
  \item \textbf{Contrato de saída importa}: sem restrição ao espaço de
        classes, \textbf{6,8\%} das respostas viram falsos erros de fraseado
  \item \textbf{O provedor de \textit{serving} importa}: o mesmo modelo
        gratuito diverge \emph{de si mesmo} entre dois serviços ($p<0{,}001$)
  \item \textbf{E0-P (prompt)}: regras de fronteira ajudam na distribuição de
        produção ($+4{,}6$ p.p., $p=0{,}012$) mas \emph{degradam} as classes
        raras ($-10{,}8$ p.p.) --- faca de dois gumes
\end{itemize}
\vspace{0.3em}
\begin{block}{Lição}
\footnotesize Cada medição de oráculo é uma fotografia da tripla
(modelo, provedor, data) --- e o instrumento move resultados por margens
maiores que várias diferenças entre modelos.
\end{block}
\end{frame}
\note{[2:30] O achado metodológico. \"A medição é parte do método.\" Os 6,8\% de falsos erros pelo contrato de saída e a divergência entre provedores (mesmo modelo diverge de si) --- isso costuma render pergunta da banca, esteja pronto. E0-P: a faca de dois gumes.}


\begin{frame}{P4 / E4 --- Robustez ao ruído do oráculo}
\begin{itemize}
  \item O ruído do oráculo vira \emph{ruído de rótulo} para o classificador
  \item E4: injeta ruído uniforme $\varepsilon \in \{0; 0{,}1; 0{,}2; 0{,}4\}$
  \item Retenção do Macro F1 final: \textbf{87\% / 74\% / 54\%} em
        $\varepsilon = 0{,}1 / 0{,}2 / 0{,}4$
  \item Desenho \emph{pessimista}: ruído real do oráculo é \textbf{estruturado}
        (categorias-irmãs), menos danoso que uniforme de mesma taxa
\end{itemize}
\vspace{0.3em}
\centering\footnotesize\color{ufprgreen}
O resultado positivo do E4 é um limite inferior do desempenho esperável com
os oráculos reais.
\end{frame}
\note{[2:00] Robustez a ruído: retenção 87/74/54\%. O ponto esperto: o desenho é PESSIMISTA (ruído uniforme), enquanto o ruído real é estruturado e menos danoso --- então o resultado é um limite inferior. Isso fortalece qualquer número positivo.}

\begin{frame}{E6 --- Entropia domina, e ``menos é mais''}
\begin{itemize}\footnotesize
  \item Entropia satura com \textbf{9,1k$\pm$0,6k} rótulos; aleatório precisa de 15,5k$\pm$1,1k
  \item Vence o aleatório em teto \textbf{8/8 sementes} (Wilcoxon $p=0{,}0078$)
  \item \textbf{Menos é mais}: treinar com a amostra ativa \emph{supera} treinar
        com o pool completo (a amostra ativa é mais balanceada)
  \item Ablação: o ganho da DRI-SL-C vem do \emph{agrupamento por predição},
        não da novidade lexical (que é contraproducente no regime contínuo)
\end{itemize}
\vspace{0.4em}
\centering\footnotesize
Um aviso operacional: \textbf{rotular mais pode piorar} a métrica macro.
\end{frame}

\begin{frame}{Decomposição da perda (por que 30\% não bastou)}
\begin{itemize}
  \item \textbf{Ruído do oráculo} (A vs.\ B, mesmos itens): custa 4,4 p.p.\ de
        acurácia, mas em Macro F1 \emph{ajuda} --- erro estruturado regulariza caudas
  \item \textbf{Seleção} (B vs.\ C): compra cobertura (620 vs.\ 493 classes), não acurácia bruta
  \item \textbf{Orçamento}: é o fator dominante --- a parada por estagnação do
        classificador leve encerra o laço cedo demais para o objetivo macro do forte
\end{itemize}
\vspace{0.4em}
\begin{block}{Correção medida}
\footnotesize Parar por \textit{proxy} do modelo forte, ou piso de $\approx$50\%
do pool. A hipótese não é infirmada: ela requer o orçamento e a parada corretos.
\end{block}
\end{frame}

\begin{frame}{Reserva --- DRI-SL em detalhe}
\footnotesize
\begin{enumerate}
  \item Codifica o pool em \textit{embeddings} de texto
  \item Agrupa por $k$-means (agrupamentos semânticos)
  \item Aloca o orçamento proporcional à massa de cada grupo
  \item Preenche cada cota gulosamente, preferindo instâncias que
        adicionam \emph{tokens} lexicais inéditos (variedade)
\end{enumerate}
\vspace{0.4em}
Determinístico, custo linear após o agrupamento, \textbf{sem usar rótulos}.
Variante \textbf{DRI-SL-C}: após o cold start, os grupos passam a ser as
classes previstas pelo classificador (para a seleção contínua).
\end{frame}
