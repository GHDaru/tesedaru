% FREEZE respeitado: todos os números são os da Seção 3.5.2 e do próprio
% apêndice (N_pop=20, 100/200 gerações, k_t=3, p_c=0,8, p_m=0,1, m_s,

\begin{tikzpicture}[
    x=1cm, y=1cm,
    caixa/.style={draw, rounded corners=2pt, align=center, text width=44mm,
                  minimum height=12mm, inner sep=3pt, font=\footnotesize},
    entrada/.style={caixa, fill=gray!8},
    medida/.style={caixa, fill=gray!8},
    produto/.style={caixa, fill=gray!14},
    seta/.style={-{Stealth[length=2.6mm]}, thick},
    rot/.style={font=\scriptsize, align=center, inner sep=2pt}
]
    % ------------------ população e proveniência da configuração -----------
    \node[entrada] (pop) at (0,0)
        {população: \mbox{$N_{pop}=20$} indivíduos;\\
         cada um: $I$ índices\\únicos do \textit{pool}};
    \node[rot, anchor=west, text width=58mm, align=left] (prov) at (2.7,0)
        {parâmetros definidos no \textit{notebook}\\
         que executou as corridas (a configuração\\
         versionada fixa apenas $|L_0|$; o padrão\\
         $0{,}7$ do cruzamento é sempre sobrescrito);\\
         a população é o único valor lido do artefato};

    % ------------------ o laço por geração ---------------------------------
    \node[caixa, text width=40mm] (torneio) at (0,-3.4)
        {seleção por torneio\\($k_t=3$)};
    \node[caixa] (cruz) at (5.3,-3.4)
        {cruzamento de um ponto\\($p_c=0{,}8$), com reparo\\de unicidade};
    \node[caixa] (mut) at (10.4,-3.4)
        {mutação (\mbox{$p_m=0{,}1$}): substitui\\
         \mbox{$m_s=\max(1,\lceil 0{,}01\cdot I\rceil)$} genes};
    \node[medida] (apt) at (10.4,-6.8)
        {aptidão medida na\\partição de aferição,\\
         \textbf{disjunta} do\\conjunto de teste};
    \node[caixa, text width=40mm] (elit) at (0,-6.8)
        {elitismo de $10\%$\\(\mbox{$N_{elite}=2$}): forma\\a nova geração};

    % ------------------ saída: anti-circularidade --------------------------
    \node[produto, text width=58mm] (reaval) at (5.3,-10.6)
        {fim do cenário: os indivíduos finais\\
         são \textbf{reavaliados no conjunto}\\
         \textbf{de teste intocado}; esse valor,\\
         tipicamente inferior ao de aptidão,\\
         é o envelope reportado};
    \node[rot, text width=44mm, anchor=north] (circ) at (10.4,-9.4)
        {otimizar e reportar na mesma partição superajustaria o envelope ao
         conjunto de avaliação (circularidade)};

    % ------------------ setas (depois de todos os nós) ----------------------
    \draw[seta] (pop) -- (torneio);
    \draw[seta, dashed] (prov.west) -- (pop.east);
    \draw[seta] (torneio) -- (cruz);
    \draw[seta] (cruz) -- (mut);
    \draw[seta] (mut) -- (apt);
    \draw[seta] (apt) -- (elit);
    \draw[seta] (elit) -- (torneio);
    % o rótulo da saída fica no vão entre o elitismo e a caixa final,
    % à direita da descida — fora do caminho e de qualquer caixa
    \draw[seta] (elit.south) |- (reaval.west);
    \node[rot, anchor=west] at (0.35,-8.9)
        {após 100 gerações (200 só no caso \mbox{$|L_0|=10$})};
    \draw[seta, dashed] (circ.north) -- (apt.south);

    % ------------------ cenários -------------------------------------------
    \node[rot, anchor=west, text width=132mm, align=left] at (-2.2,-12.6)
        {4 cenários por tamanho (maximização e minimização de Acurácia e de
         Macro F1) $\times$ 10 tamanhos de $L_0$, de 10 a 30.000.};
\end{tikzpicture}
